cmake_minimum_required(VERSION 3.21)
project(TrafficStatisticsTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Concurrent Gui Qml Quick)
add_executable(trafficStatisticsTests main.cpp stubs/vpnConnection.h
    ../../client/ui/controllers/trafficStatisticsController.cpp
    ../../client/ui/controllers/trafficStatisticsController.h)
target_include_directories(trafficStatisticsTests PRIVATE stubs ../../client/ui/controllers)
target_link_libraries(trafficStatisticsTests PRIVATE Qt6::Core Qt6::Concurrent iphlpapi)
set(CMAKE_AUTORCC ON)
add_executable(trafficStatisticsUiSmoke uiSmoke.cpp ../../client/images/images.qrc)
target_link_libraries(trafficStatisticsUiSmoke PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick)
get_filename_component(TEST_REPO_PATH "../.." ABSOLUTE)
target_compile_definitions(trafficStatisticsUiSmoke PRIVATE REPO_PATH="${TEST_REPO_PATH}" TEST_OUTPUT_PATH="${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
add_test(NAME trafficStatistics COMMAND trafficStatisticsTests)
add_test(NAME trafficStatisticsUi COMMAND trafficStatisticsUiSmoke -platform offscreen)

add_executable(trafficNavigationSmoke navigationSmoke.cpp ../../client/ui/utils/pageEnum.h ../../client/images/images.qrc)
target_include_directories(trafficNavigationSmoke PRIVATE ../../client)
target_link_libraries(trafficNavigationSmoke PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick)
target_compile_definitions(trafficNavigationSmoke PRIVATE REPO_PATH="${TEST_REPO_PATH}" TEST_OUTPUT_PATH="${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME trafficNavigation COMMAND trafficNavigationSmoke -platform offscreen)
