cmake_minimum_required(VERSION 3.21)
project(AwgProxyTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(Qt6 REQUIRED COMPONENTS Network)
add_executable(proxySnapshotTests snapshot.cpp)
target_include_directories(proxySnapshotTests PRIVATE ../../client)
target_link_libraries(proxySnapshotTests PRIVATE Qt6::Core Qt6::Network)
add_executable(systemProxyTests main.cpp ../../client/core/proxy/windowsSystemProxyManager.cpp)
target_include_directories(systemProxyTests PRIVATE ../../client)
target_link_libraries(systemProxyTests PRIVATE Qt6::Core)
if(WIN32)
    target_link_libraries(systemProxyTests PRIVATE wininet)
endif()
enable_testing()
add_test(NAME systemProxyRecovery COMMAND systemProxyTests)
add_test(NAME proxySnapshot COMMAND proxySnapshotTests)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Qml)
add_executable(proxyTestHelper fakeHelper.cpp)
target_link_libraries(proxyTestHelper PRIVATE Qt6::Core)
add_executable(proxyLifecycleTests lifecycle.cpp
    ../../client/core/proxy/proxyConnection.cpp ../../client/core/proxy/proxyConnection.h
    ../../client/core/proxy/windowsSystemProxyManager.cpp)
target_include_directories(proxyLifecycleTests PRIVATE stubs ../../client)
target_link_libraries(proxyLifecycleTests PRIVATE Qt6::Core Qt6::Network Qt6::Qml)
if(WIN32)
    target_link_libraries(proxyLifecycleTests PRIVATE wininet)
endif()
add_dependencies(proxyLifecycleTests proxyTestHelper)
add_test(NAME proxyLifecycle COMMAND proxyLifecycleTests)
find_package(Qt6 REQUIRED COMPONENTS Gui Qml Quick)
set(CMAKE_AUTORCC ON)
add_executable(proxySettingsUi uiSmoke.cpp ../../client/images/images.qrc)
target_link_libraries(proxySettingsUi PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick)
get_filename_component(TEST_REPO_PATH "../.." ABSOLUTE)
target_compile_definitions(proxySettingsUi PRIVATE REPO_PATH="${TEST_REPO_PATH}" TEST_OUTPUT_PATH="${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME proxySettingsUi COMMAND proxySettingsUi -platform offscreen)
